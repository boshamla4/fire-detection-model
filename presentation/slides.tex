\section{Motivation}
% ═══════════════════════════════════════════════════════════════════════════

\begin{frame}{Why Forest Fire Detection?}
  \begin{columns}
    \column{0.55\textwidth}
      \begin{itemize}
        \item \textbf{13.4M hectares} burned in 2023 (Europe, North Africa, North America)
        \item Economic and ecological losses worth billions annually
        \item \textbf{Critical detection window:} early minutes determine total area burned
      \end{itemize}
      \vspace{1em}
      \textbf{Existing systems fail because:}
      \begin{itemize}
        \item Satellite revisit: \textbf{30 min–several hours} delay
        \item Ground sensors: limited coverage, costly deployment
        \item Cloud AI: works, but needs \textbf{reliable network} — unavailable in forests
      \end{itemize}
    \column{0.45\textwidth}
      % \includegraphics[width=\textwidth]{figures/wildfire_photo.jpg}
      \begin{block}{Target latency}
        \centering\Large $< 5$ seconds end-to-end
      \end{block}
  \end{columns}
\end{frame}

\section{System Architecture}
% ═══════════════════════════════════════════════════════════════════════════

\begin{frame}{Edge-Cloud Architecture Overview}
  \begin{columns}
    \column{0.5\textwidth}
      \textbf{Three layers:}
      \begin{enumerate}
        \item \textbf{UAV Edge Layer} — YOLOv11n inference onboard
        \item \textbf{Communication Layer} — 4G/LTE · WiFi · LoRa fallback
        \item \textbf{Cloud Layer} — event aggregation, fire map, alerts
      \end{enumerate}
      \vspace{0.7em}
      \textbf{Three connectivity modes:}
      \begin{itemize}
        \item \textcolor{green!60!black}{Connected} — stream events live
        \item \textcolor{orange}{Zone blanche} — buffer locally
        \item \textcolor{red}{Critical} — LoRa emergency packet
      \end{itemize}
    \column{0.5\textwidth}
      % \includegraphics[width=\textwidth]{figures/architecture.png}
      \begin{block}{Key insight}
        Only~\textbf{500-byte JSON events} are transmitted\\
        — not raw video frames.\\
        Works even over LoRa (5 kbps).
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Communication Link Comparison}
  \begin{table}
    \centering
    \begin{tabular}{lccc}
      \toprule
      \textbf{Link} & \textbf{Bandwidth} & \textbf{Latency} & \textbf{Use case} \\
      \midrule
      4G/LTE       & $\sim$10 Mbps & 20–50 ms  & Normal operation \\
      WiFi 802.11n & $\sim$50 Mbps & 5–15 ms   & Near base station \\
      LoRa 868 MHz & $\sim$5 kbps  & 500 ms    & Zone blanche \\
      \bottomrule
    \end{tabular}
  \end{table}
\end{frame}

\section{Dataset \& Methodology}
% ═══════════════════════════════════════════════════════════════════════════

\begin{frame}{Dataset: D-Fire}
  \begin{columns}
    \column{0.5\textwidth}
      \begin{itemize}
        \item \textbf{21,527 images} of fire and smoke
        \item Classes: \texttt{fire}, \texttt{smoke}
        \item Split: 70 / 20 / 10 (train / val / test)
        \item Diverse sources: cameras, drones, online
        \item Publicly available on GitHub (gaiasd/DFireDataset)
      \end{itemize}
    \column{0.5\textwidth}
      % \includegraphics[width=\textwidth]{figures/dataset_samples.png}
      \begin{block}{Preprocessing}
        640×640 resize · Mosaic augmentation\\
        Random affine · Histogram equalization
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{YOLOv11 — Model Choice}
  \begin{columns}
    \column{0.55\textwidth}
      \textbf{Why YOLOv11?}
      \begin{itemize}
        \item Released September 2024 (state-of-the-art)
        \item C3k2 + C2PSA backbone — better small-object detection
        \item \textbf{22\% fewer parameters} than YOLOv8m, same mAP
        \item Exports to ONNX, TensorRT, CoreML for edge devices
      \end{itemize}
      \vspace{0.5em}
      \textbf{Two variants evaluated:}
      \begin{itemize}
        \item \texttt{YOLOv11x} — cloud GPU baseline
        \item \texttt{YOLOv11n} + INT8 — \textbf{edge deployment}
      \end{itemize}
    \column{0.45\textwidth}
      \begin{block}{Training config}
        \texttt{epochs=100, batch=16}\\
        \texttt{optimizer=AdamW, lr=0.01}\\
        \texttt{pretrained=COCO}\\
        Full fine-tuning (all layers)
      \end{block}
  \end{columns}
\end{frame}
