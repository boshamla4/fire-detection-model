\section{Motivation}
% ═══════════════════════════════════════════════════════════════════

\begin{frame}{Why Real-Time Forest Fire Detection?}
  \begin{columns}
    \column{0.55\textwidth}
      \begin{itemize}
        \item \textbf{13.4M hectares} burned in 2023 globally
        \item Billions in economic and ecological losses annually
        \item Early detection is critical — every minute matters
      \end{itemize}
      \vspace{0.6em}
      \textbf{Why existing systems fail:}
      \begin{itemize}
        \item Satellites: \textbf{30 min – several hours} revisit delay
        \item Ground sensors: limited coverage, costly deployment
        \item Cloud-only AI: breaks down in \textbf{zone blanche} — remote
              forests with no connectivity
      \end{itemize}
    \column{0.45\textwidth}
      \begin{block}{\centering Operational target}
        \centering\Large $< 5$ seconds\\end-to-end detection latency
      \end{block}
      \vspace{0.5em}
      \begin{alertblock}{\centering Our answer}
        \centering AI onboard the UAV\\
        Works even without network
      \end{alertblock}
  \end{columns}
\end{frame}

    measures end-to-end operational latency.
  \end{alertblock}
\end{frame}

      \begin{itemize}

      \end{itemize}

      \begin{itemize}
        \item 22\% fewer parameters
        \item C3k2 + C2PSA attention
        \item Comparable mAP
      \end{itemize}
  \end{columns}
